name: Build APK

on:
  push:
    branches: [main, master]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Arrange files
        run: bash setup.sh
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17
      - uses: gradle/actions/setup-gradle@v3
        with:
          gradle-version: 8.5
      - name: Build release APK
        run: gradle assembleRelease --no-daemon
      - name: Rename APK
        run: cp app/build/outputs/apk/release/app-release.apk ValyLauncher.apk
      - uses: actions/upload-artifact@v4
        with:
          name: ValyLauncher-apk
          path: ValyLauncher.apk
      - name: Publish release
        uses: softprops/action-gh-release@v2
        with:
          tag_name: build-${{ github.run_number }}
          name: Valy Launcher build ${{ github.run_number }}
          files: ValyLauncher.apk
